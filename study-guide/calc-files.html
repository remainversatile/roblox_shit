EXAM2 Calculator Files

The EXAM2 program in seven small Python files, so it fits in the calculator's Python memory. Same menus and reference cards as before, laid out like ECON2.

Why it's seven files now

The single 42 KB EXAM2.py stopped with MemoryError: memory allocation failed on line 1, before it ran. The calculator runs out of memory while compiling a file that size, and ECON2 (24 KB) is about the most it handles. EXAM2.py is now a 2.7 KB launcher. Each helper loads only while you're using it and is removed from memory afterward, so only about three files are in memory at once.

Put it on the calculator

  1. Delete the old EXAM2 from the calculator first.
  2. Save all seven files with the names exactly as shown (EXAM2, E2U, E2P, E2I, E2Z, E2C, E2X). Use the raw-file links, the copy buttons, or the files from the chat.
  3. Open TI Connect CE, connect the calculator, and drag all seven .py files onto it.
  4. On the calculator, open the Python app, select EXAM2, and run it. Don't run the E2 files directly. They're helpers.
  5. If a menu says “Send E2__.py to the calculator too,” that file didn't transfer.

Still out of memory? Quit the Python app and reopen it so memory starts empty, then run EXAM2 first. The program also catches memory errors during use and sends you back to the menu instead of crashing.

Using it

  • Main menu: 1 Reference cards · 2 Probability (x, x̄, p̄) · 3 Percent → cutoff · 4 Critical z / t · 5 Confidence interval · 6 Minimum sample size · 7 CI parts / std dev.
  • Reference cards: 8 color-coded topics, 82 cards: z-score sheet, formula sheet, Chapters 6–8, mix-ups, calculator vs. table, concepts and exam rules.
  • Paging: enter next, 1 previous, 2 jump to a page, 0 menu. Press clear to leave a drawn card. A blank entry cancels a calculation.
  • Answers: TABLE is the course z-table method and matches the answer key. EXACT keeps full precision. When they differ, enter TABLE.

Tested on a computer against every Practice Test 2 answer, with a stand-in for the calculator's ti_draw screen, in both graphics and text mode, plus 1,500 random-input runs. It hasn't been run on a physical TI-84 Evo. Ask Professor Spencer before keeping it on the calculator during the exam.

#   EXAM2 E2U E2P E2I E2Z E2C E2X  (.py)
# Helpers load only while in use, to save memory.
import sys
from E2U import *

# ---- reference cards (from E2Z, E2C, E2X) ----
# Each module has get(i) -> list of
# "Title|line\nline|label;numerator;denominator" strings.
MODS = [("E2Z", 0), ("E2Z", 1), ("E2C", 0), ("E2C", 1), ("E2C", 2), ("E2X", 0), ("E2X", 1), ("E2X", 2)]
NAMES = ["Z-score sheet", "Formula sheet", "Ch 6 normal", "Ch 7 sampling", "Ch 8 intervals",
         "Mix-ups", "Calculator / table", "Concepts / exam"]

def imp(name):
  try:
    return __import__(name)
  except ImportError:
    raise ValueError("Send " + name + ".py to the calculator too")

def free(name):
  # drop a helper module so its memory comes back
  try:
    del sys.modules[name]
  except Exception:
    pass
  gc.collect()

def run(name, fn):
  m = imp(name)
  try:
    getattr(m, fn)()
  finally:
    del m
    free(name)

def load(i):
  name, j = MODS[i]
  m = imp(name)
  raw = m.get(j)
  del m
  free(name)
  cs = []
  for r in raw:
    p = r.split("|")
    cs += mk(p[0], p[1], tuple(p[2].split(";")) if len(p) > 2 else None)
  del raw
  gc.collect()
  return cs

def refs():
  while True:
    disp_clr()
    print("--- REFERENCE CARDS ---")
    for i in range(8):
      print(str(i + 1) + " " + NAMES[i])
    print("0 Main menu")
    s = input("> ").strip()
    if s == "0":
      return
    if len(s) == 1 and s in "12345678":
      i = int(s) - 1
      cs = load(i)
      ACC[0] = COLS[i]
      try:
        browse(cs)
      finally:
        ACC[0] = NAVY
        del cs
        gc.collect()

def main():
  while True:
    disp_clr()
    print("=== EXAM2: CH 6-8 ===")
    print("1 Reference cards")
    print("2 Probability x/xbar/pbar")
    print("3 Percent -> cutoff")
    print("4 Critical z / t")
    print("5 Confidence interval")
    print("6 Minimum sample size")
    print("7 CI parts / std dev")
    print("0 Exit")
    s = input("> ").strip()
    if s == "0":
      break
    try:
      if s == "1":
        refs()
      elif s in ["2", "3", "4"]:
        run("E2P", ["prob", "cutoff", "crit"][int(s) - 2])
      elif s in ["5", "6", "7"]:
        run("E2I", ["ci", "size", "more"][int(s) - 5])

    except MemoryError:
      disp_clr()
      print("Out of memory. Close other")
      print("apps, then rerun EXAM2.")
      input("ENTER: menu")
    gc.collect()

# E2U - shared math, input and screen code for EXAM2.
# Keep with EXAM2.py. Plain ASCII, math + ti_draw.
# TABLE = course z-table method (matches key).
# EXACT = full calculator precision.
import math
import gc
try:
  import ti_draw as dr
  GR = True
except ImportError:
  GR = False
try:
  from ti_system import disp_clr
except ImportError:
  def disp_clr():
    print("\n" * 3)

def parse(s):
  for c in s.replace("sqrt", ""):
    if c not in "0123456789.+-*/() ":
      raise ValueError("Use digits, / and sqrt()")
  try:
    return float(eval(s, {"sqrt": math.sqrt}))
  except ZeroDivisionError:
    raise ValueError("Zero denominator")
  except Exception:
    raise ValueError("Not a number")

def rnd(x, d):
  m = 10 ** d
  g = -1 if x < 0 else 1
  return g * math.floor(abs(x) * m + 0.5 + 1e-9) / m

def f4(x):
  return "%.4f" % x

def f2(x):
  return "%.2f" % x

def sh(x):
  return "%.6g" % x

# ---- normal and t ----
try:
  erf = math.erf
except AttributeError:
  def erf(x):
    g = -1 if x < 0 else 1
    x = abs(x)
    if x > 6:
      return g
    t = s = x
    k = 0
    while t > 1e-16 * s:
      k += 1
      t = t * 2 * x * x / (2 * k + 1)
      s += t
    return g * 1.1283791670955126 * math.exp(-x * x) * s

def phi(z):
  return 0.5 * (1 + erf(z / 1.4142135623730951))

def tphi(z):
  z = rnd(z, 2)
  if z < -3.49:
    return 0.0
  if z > 3.49:
    return 1.0
  return rnd(phi(z), 4)

def bis(f, a, lo, hi):
  # x in [lo,hi] where increasing f(x) = a
  for i in range(48):
    m = (lo + hi) / 2
    if f(m) < a:
      lo = m
    else:
      hi = m
  return (lo + hi) / 2

def invphi(a):
  return bis(phi, a, -9.0, 9.0)

def tablez(a):
  # closest 4-place table area; a tie takes the midpoint
  z0 = rnd(invphi(a), 2)
  best = 9
  zs = []
  for i in range(-3, 4):
    z = rnd(z0 + i / 100, 2)
    d = abs(rnd(phi(z), 4) - a)
    if d < best - 1e-9:
      best = d
      zs = [z]
    elif abs(d - best) <= 1e-9:
      zs.append(z)
  return rnd(sum(zs) / len(zs), 3)

def tcdf(t, v):
  # P(T < t) for t >= 0, whole df (exact series)
  th = math.atan(t / math.sqrt(v))
  c2 = math.cos(th) ** 2
  w = 1.0
  tot = 1.0
  if v % 2:
    for k in range(1, (v - 1) // 2):
      w *= c2 * 2 * k / (2 * k + 1)
      tot += w
    a = th + (math.sin(th) * math.cos(th) * tot if v > 1 else 0)
    a = 2 * a / math.pi
  else:
    for k in range(1, v // 2):
      w *= c2 * (2 * k - 1) / (2 * k)
      tot += w
    a = math.sin(th) * tot
  return (1 + a) / 2

def invt(up, v):
  # positive t with upper-tail area up
  if v > 200:
    z = invphi(1 - up)
    return z + (z ** 3 + z) / (4 * v) + (5 * z ** 5 + 16 * z ** 3 + 3 * z) / (96 * v * v)
  return bis(lambda t: tcdf(t, v), 1 - up, 0.0, 1000.0)

TZ = {80: 1.28, 90: 1.645, 95: 1.96, 98: 2.33, 99: 2.576}

def zcrit(c):
  up = 1 - (1 - c / 100) / 2
  k = int(c + 0.5)
  if k in TZ and abs(c - k) < 1e-9:
    return TZ[k], invphi(up)
  return tablez(up), invphi(up)

# ---- input ----
def ask(label, pos=False, nonneg=False):
  while True:
    disp_clr()
    print(label)
    print("Number, a/b, or sqrt()")
    print("Blank = cancel")
    s = input("> ").strip()
    if s == "":
      raise ValueError("Cancelled")
    try:
      x = parse(s)
      if pos and x <= 0:
        raise ValueError("Must be positive")
      if nonneg and x < 0:
        raise ValueError("Cannot be negative")
      return x
    except ValueError as e:
      print(str(e))
      input("ENTER to retry")

def askp(label):
  p = ask(label + " (.66 or 66)", True)
  if p > 1:
    p /= 100
  if p >= 1:
    raise ValueError("Need 0 < p < 1")
  return p

def askc():
  c = ask("Confidence % (95 or .95)", True)
  if c < 1:
    c *= 100
  if c >= 100:
    raise ValueError("Need under 100%")
  return c

def pick(lines, n):
  disp_clr()
  for s in lines:
    print(s)
  k = input("> ").strip()
  return int(k) if k in [str(i) for i in range(1, n + 1)] else 0

# ---- screen and cards (ECON2 layout) ----
def wrap(t, w=28):
  out = []
  line = ""
  for word in t.split():
    if len(word) > w:
      word = word[:w]
    if line and len(line) + len(word) + 1 > w:
      out.append(line)
      line = word
    else:
      line = word if not line else line + " " + word
  if line:
    out.append(line)
  return out

def ink(c):
  dr.set_color(c[0], c[1], c[2])

NAVY = (10, 65, 100)
COLS = [(10, 65, 100), (0, 105, 75), (150, 75, 0), (130, 35, 110),
        (165, 35, 45), (55, 60, 135), (0, 110, 130), (110, 85, 20)]
ACC = [NAVY]

def tint(c):
  return (c[0] + (255 - c[0]) * 7 // 8, c[1] + (255 - c[1]) * 7 // 8, c[2] + (255 - c[2]) * 7 // 8)

def box(x, y, w, h):
  try:
    dr.fill_rect(x, y, w, h)
  except Exception:
    pass

def frame(title, tag, i=0, n=0):
  a = ACC[0]
  dr.clear()
  dr.set_pen("thin", "solid")
  ink(tint(a))
  box(0, 0, 320, 24)
  ink(a)
  dr.draw_text(8, 5, title)
  box(0, 24, 320, 2)
  if n > 1:
    ink((205, 210, 215))
    box(0, 26, 320, 3)
    ink(a)
    box(0, 26, 320 * i // n, 3)
  ink((232, 235, 238))
  box(0, 188, 320, 22)
  ink((70, 70, 70))
  dr.draw_text(8, 192, tag)

def mk(title, body, fr=None):
  # card = (title, lines, fraction); overflow goes to extra pages
  flat = []
  for s in (body.split("\n") if isinstance(body, str) else body):
    if s[:1] == "!":
      flat += ["!" + w for w in wrap(s[1:], 27)]
    else:
      flat += wrap(s) if s else [""]
  room = 4 if fr else 8
  out = [(title, flat[:room], fr)]
  flat = flat[room:]
  while flat:
    out.append((title, flat[:8], None))
    flat = flat[8:]
  return out

def show(cd, i=1, n=1):
  title, lines, fr = cd
  foot = "Page %d of %d   CLEAR = nav" % (i, n) if n > 1 else "Press CLEAR"
  if GR:
    a = ACC[0]
    frame(title, foot, i, n)
    y = 35
    if fr:
      ink(tint(a))
      box(4, 31, 312, 74)
      ink(a)
      dr.draw_text(10, 33, fr[0] + " =")
      ink((25, 25, 25))
      dr.draw_text(24, 52, fr[1])
      ink(a)
      dr.draw_line(20, 73, 300, 73)
      dr.draw_line(20, 74, 300, 74)
      ink((25, 25, 25))
      dr.draw_text(24, 79, fr[2])
      y = 112
    for s in lines:
      if s[:1] == "!":
        ink(tint(a))
        box(4, y - 2, 312, 19)
        ink(a)
        box(4, y - 2, 3, 19)
        ink((0, 0, 0))
        dr.draw_text(12, y, s[1:])
      else:
        ink(a if s[-1:] == ":" else (25, 25, 25))
        dr.draw_text(8, y, s)
      y += 18

  else:
    disp_clr()
    print("== " + title + " ==")
    if fr:
      print(fr[0] + " =")
      print("  " + fr[1])
      print(" " + "-" * 26)
      print("  " + fr[2])
    for s in lines:
      print(">> " + s[1:] if s[:1] == "!" else s)

def browse(cards):
  i = 0
  n = len(cards)
  while True:
    show(cards[i], i + 1, n)
    disp_clr()
    print("ENTER next | 1 previous")
    print("2 jump to page | 0 menu")
    c = input("> ").strip()
    if c == "0":
      return
    if c == "1":
      i = (i - 1) % n
    elif c == "2":
      try:
        j = int(input("Page 1-%d: " % n))
        if 1 <= j <= n:
          i = j - 1
      except ValueError:
        pass
    else:
      i = (i + 1) % n

def note(title, body):
  for cd in mk(title, body):
    show(cd)
    if not GR:
      input("ENTER: continue")

# E2P - probability, cutoff and critical values for EXAM2.
# Loaded only while in use. Keep with EXAM2.py.
from E2U import *

# ---- probability (Ch 6, Ch 7) ----
RULE = ["", "left area", "1 - left area", "left(b) - left(a)", "1 - between"]
WHAT = ["", "Below a", "Above a", "Between a and b", "Outside a to b"]

def area(pa, pb, k):
  return [0, pa, 1 - pa, pb - pa, 1 - pb + pa][k]

def prob():
  w = pick(["1 One value x (Ch 6)", "2 Sample mean xbar (Ch 7)", "3 Sample prop pbar (Ch 7)", "0 Cancel"], 3)
  if not w:
    return
  if w == 3:
    c = askp("Population proportion p")
    n = ask("Sample size n", True)
    se = math.sqrt(c * (1 - c) / n)
    st = rnd(se, 4)
    ok = n * c >= 5 and n * (1 - c) >= 5
    cs = mk("Setup: sample prop", ["np = " + f2(n * c), "n(1-p) = " + f2(n * (1 - c)),
            "Both >= 5: normal" if ok else "NOT both >= 5!", "!SE = " + f4(se)],
            ("SE", "sqrt[ " + sh(c) + "(" + sh(1 - c) + ") ]", "sqrt(" + sh(n) + ")"))
  else:
    c = ask("Population mean mu")
    sg = ask("Population sigma", True)
    n = ask("Sample size n", True) if w == 2 else 1
    se = st = sg / math.sqrt(n)
    if w == 2:
      cs = mk("Setup: sample mean", ["E(xbar) = mu = " + sh(c), "!SE = " + f4(se),
              "n >= 30: approx normal" if n >= 30 else "n < 30: pop must be normal"],
              ("SE", sh(sg), "sqrt(" + sh(n) + ")"))
    else:
      cs = mk("Setup: one value", ["Center mu = " + sh(c), "Spread sigma = " + sh(sg)], ("z", "x - mu", "sigma"))
  k = pick(["1 Below a | 2 Above a", "3 Between | 4 Outside", "5 Within +/-d of center", "0 Cancel"], 5)
  if not k:
    return
  if k == 5:
    d = ask("Distance d from center", True)
    a, b, k = c - d, c + d, 3
  elif k > 2:
    a = ask("Lower value a")
    b = ask("Upper value b")
    if b < a:
      a, b = b, a
  else:
    a = b = ask("Value a")
  za, zb = (a - c) / se, (b - c) / se
  ta, tb = rnd((a - c) / st, 2), rnd((b - c) / st, 2)
  for v, z, t in (("a", za, ta), ("b", zb, tb))[:1 if k < 3 else 2]:
    cs += mk("z for " + v, ["z = " + f4(z), "!Table z = " + f2(t), "Left area = " + f4(tphi(t))],
             ("z", sh(a if v == "a" else b) + " - " + sh(c), f4(st)))
  pt = area(tphi(ta), tphi(tb), k)
  ans = [WHAT[k], "Rule: " + RULE[k], "!TABLE P = " + f4(pt), "EXACT P = " + f4(area(phi(za), phi(zb), k)),
         "Percent: " + f2(pt * 100) + "%"]
  if k > 2:
    ans.append("Subtract AREAS, not z.")
  browse(cs + mk("Answer", ans))
  graph(k, ta, tb, "P = " + f4(pt))

# ---- percent -> cutoff ----
def cutoff():
  k = pick(["1 Bottom __% (below)", "2 Top __% (above)", "3 Middle __%", "0 Cancel"], 3)
  if not k:
    return
  p = askp("Percent")
  L = [p] if k == 1 else ([1 - p] if k == 2 else [(1 - p) / 2, (1 + p) / 2])
  g = pick(["Solve for:", "1 Cutoff, one x", "2 Cutoff, xbar", "3 Cutoff, pbar", "4 Unknown mean mu", "0 Cancel"], 4)
  if not g:
    return
  if g == 3:
    c = askp("Population proportion p")
    n = ask("Sample size n", True)
    se = math.sqrt(c * (1 - c) / n)
    st = rnd(se, 4)
  else:
    c = ask("Known value x" if g == 4 else "Population mean mu")
    sg = ask("Population sigma", True)
    se = st = sg / math.sqrt(ask("Sample size n", True)) if g == 2 else sg
  cs = []
  for a in L:
    zt, ze = tablez(a), invphi(a)
    cs += mk("LEFT area " + f4(a), ["Find " + f4(a) + " INSIDE table.", "!Table z = " + str(zt), "Calculator z = " + f4(ze)])
    if g == 4:
      cs += mk("Unknown mean", ["mu = x - (z)(sigma)", "= " + sh(c) + " - (" + str(zt) + ")(" + sh(st) + ")",
               "!TABLE mu = " + f4(c - zt * st), "EXACT mu = " + f4(c - ze * se)])
    else:
      cs += mk("Cutoff", ["= center + (z)(SE)", "= " + sh(c) + " + (" + str(zt) + ")(" + sh(st) + ")",
               "!TABLE = " + f4(c + zt * st), "EXACT = " + f4(c + ze * se)])
  browse(cs)
  z1 = tablez(L[0])
  graph(k, z1, tablez(L[-1]), ["", "Bottom ", "Top ", "Middle "][k] + f2(p * 100) + "%")

# ---- critical values ----
def crit():
  c = askc()
  al = 1 - c / 100
  tb, ex = zcrit(c)
  cs = mk("Critical probabilities", ["Confidence " + sh(c) + "%", "alpha = " + f4(al), "!alpha/2 = " + f4(al / 2),
          "!1 - alpha/2 = " + f4(1 - al / 2)])
  cs += mk("Critical z", ["!Table z = +/-" + str(tb), "Calculator z = " + f4(ex), "sigma known, proportions,", "sample size."])
  disp_clr()
  s = input("n for a t value (ENTER skip): ").strip()
  if s:
    v = int(parse(s) - 1 + 0.5)
    if v >= 1:
      cs += mk("Critical t", ["df = n - 1 = %d" % v, "Upper area " + f4(al / 2), "!t = +/-%.3f" % invt(al / 2, v), "Only s given."])
  browse(cs)

# E2I - intervals, sample size and CI tools for EXAM2.
# Loaded only while in use. Keep with EXAM2.py.
from E2U import *

# ---- confidence intervals ----
def ival(est, et, ee, what):
  d = min(int(ask("Decimals the question wants", nonneg=True)), 6)
  fm = "%." + str(d) + "f"
  return mk("Interval", ["!TABLE (" + fm % (est - et) + ", " + fm % (est + et) + ")",
            "EXACT (" + fm % (est - ee) + ", " + fm % (est + ee) + ")", "Width = 2E = " + fm % (2 * et),
            "We are __% confident the", what, "is in this interval."])

def ci():
  k = pick(["1 mu, sigma known (z)", "2 mu, only s given (t)", "3 p from pbar", "4 p from count x", "0 Cancel"], 4)
  if not k:
    return
  if k < 3:
    xb = ask("Sample mean xbar")
    sg = ask("Population sigma" if k == 1 else "Sample s", True)
    n = ask("Sample size n", True)
    c = askc()
    se = sg / math.sqrt(n)
